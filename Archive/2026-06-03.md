# New Shorthand
I’ll propose an existing, easy shorthand syntax you can type in Sublime, then map it to a lightweight Python-based Sublime package flow that transforms and replaces selected text into either ASCII tree or box-diagram output. I’m going to first outline a minimal grammar and show concrete input/output examples that fit your sample.

Use an existing pattern: an indented outline DSL (already familiar from Markdown lists, YAML, Org-mode).  
It is fast to type, visually readable, and trivial to parse.

Recommended shorthand
1. Root node: `Title --> state_tag`  
2. Child node: two-space indent per level  
3. Optional note: `field_name :: note text`  
4. Keep labels human-readable; script normalizes leaf labels to `snake_case`  
5. Ignore uncommon edge cases for now (exactly your request)

Example input you can type quickly
```text
PBI Dashboard --> current_state
  business domain leader
  business owner emaill
  top user l30d
test
  test2
```

Minimal Sublime package flow
1. Write shorthand in Sublime.
2. Run a command that calls a tiny Python parser/renderer.
3. Replace selected text with output (`type0` tree or `type1` box view).

Sublime plugin code (single file)
```python
import re
import sublime
import sublime_plugin


DESC = {
    "business_domain_leader": "canonical mapping to data domain owner (domain accountability)",
    "business_owner_email": "curated ownership attribution (experiential / steward-enriched mapping)",
    "top_user_l30d": "highest activity user based on L30D snapshot (as of 2026-06-02)",
}


def snake(s: str) -> str:
    s = s.strip().lower()
    s = re.sub(r"[^a-z0-9]+", "_", s)
    s = re.sub(r"_+", "_", s).strip("_")
    # tiny typo helper for your sample
    if s == "business_owner_emaill":
        s = "business_owner_email"
    return s


def parse_outline(text: str):
    roots = []
    stack = []  # (level, node)

    for raw in text.splitlines():
        if not raw.strip():
            continue

        indent = len(raw) - len(raw.lstrip(" "))
        level = indent // 2
        line = raw.strip()

        state = None
        if "-->" in line:
            left, right = line.split("-->", 1)
            line = left.strip()
            state = right.strip()

        node = {"label": line, "state": state, "children": []}

        while stack and stack[-1][0] >= level:
            stack.pop()

        if stack:
            stack[-1][1]["children"].append(node)
        else:
            roots.append(node)

        stack.append((level, node))

    return roots


def render_type0(roots):
    lines = []

    def walk(children, prefix=""):
        for i, c in enumerate(children):
            last = i == len(children) - 1
            branch = "└─" if last else "├─"

            key = snake(c["label"]) if not c["children"] else c["label"]
            line = f"{prefix}{branch} {key}"

            desc = DESC.get(snake(c["label"]))
            if desc:
                line += f"   -- {desc}"
            lines.append(line)

            child_prefix = prefix + ("   " if last else "|  ")
            walk(c["children"], child_prefix)

    for r_i, r in enumerate(roots):
        if r_i > 0:
            lines.append("")

        title = r["label"]
        if r["state"]:
            title += f" ({r['state'].replace('_', ' ').title()})"
        lines.append(title)
        walk(r["children"])

    return "\n".join(lines)


def render_type1(roots):
    # lightweight "box map": top root + first-level children
    if not roots:
        return ""

    root = roots[0]
    kids = root["children"]

    root_title = root["label"]
    w = max(30, len(root_title) + 4)
    top = " " * 16 + "┌" + "─" * (w - 2) + "┐"
    mid = " " * 16 + "│ " + root_title.ljust(w - 4) + " │"
    bot = " " * 16 + "└" + "─" * (w - 2) + "┘"
    stem = " " * (16 + w // 2) + "│"

    boxes = []
    bw = 28
    for k in kids:
        t = k["label"][:bw - 4]
        b1 = "┌" + "─" * (bw - 2) + "┐"
        b2 = "│ " + t.ljust(bw - 4) + " │"
        b3 = "└" + "─" * (bw - 2) + "┘"
        boxes.append((b1, b2, b3))

    if not boxes:
        return "\n".join([top, mid, bot])

    spacer = "  "
    row1 = spacer.join([b[0] for b in boxes])
    row2 = spacer.join([b[1] for b in boxes])
    row3 = spacer.join([b[2] for b in boxes])

    connector = " " * 8 + "┌" + "─" * max(6, len(row1) - 18) + "┐"

    return "\n".join([top, mid, bot, stem, connector, row1, row2, row3])


class QuickOutlineTransformCommand(sublime_plugin.TextCommand):
    def run(self, edit, output_type="type0"):
        sels = sorted(list(self.view.sel()), key=lambda r: r.begin(), reverse=True)

        for region in sels:
            if region.empty():
                region = sublime.Region(0, self.view.size())

            source = self.view.substr(region)
            roots = parse_outline(source)

            if output_type == "type1":
                out = render_type1(roots)
            else:
                out = render_type0(roots)

            self.view.replace(edit, region, out)
```
